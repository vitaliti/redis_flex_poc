#include "lru_eviction_policy.hpp"

void LruEvictionPolicy::print() const
{
    printf("Keys in RAM/CACHE \n");
    for (const auto& key : m_lru)
    {
        printf("Key:%s \n", key.c_str());
    }
}

void LruEvictionPolicy::updateEvictionCandidate(const std::string& key, Entry& entry, const bool isInserted)
{
    if (!isInserted)
    {
        m_lru.erase(entry.m_lruIt);
    }

    m_lru.push_back(key);
    entry.m_lruIt = std::prev(m_lru.end());
}

void LruEvictionPolicy::remove(const Entry& entry)
{
    m_lru.erase(entry.m_lruIt);
}

std::optional<std::string> LruEvictionPolicy::getEvictionCandidate()
{
    if (m_lru.empty())
    {
        return std::nullopt;
    }

    return m_lru.front();
}
